cmake_minimum_required(VERSION 3.25)
project(nv_safety_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
set(NEUROVECTOR_DRONE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../drone" CACHE PATH "Drone source checkout")
set(NEUROVECTOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory("${NEUROVECTOR_DRONE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/drone" EXCLUDE_FROM_ALL)
rosidl_generate_interfaces(${PROJECT_NAME}
  msg/Box.msg msg/Coverage.msg msg/Obstacle.msg msg/LandingSurface.msg
  msg/Window.msg msg/WifiPath.msg msg/DeliveryObservation.msg msg/SafetyDecision.msg
  DEPENDENCIES geometry_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_library(snapshot STATIC src/snapshot.cpp)
target_include_directories(snapshot PUBLIC src)
target_link_libraries(snapshot nv_avoidance "${typesupport}")
ament_target_dependencies(snapshot geometry_msgs)
add_executable(safety_bridge src/safety_bridge.cpp src/px4_state.cpp src/command_channel.cpp)
target_link_libraries(safety_bridge snapshot)
ament_target_dependencies(safety_bridge rclcpp px4_msgs)
add_executable(bridge_integration_test test/bridge_integration_test.cpp)
target_link_libraries(bridge_integration_test snapshot)
ament_target_dependencies(bridge_integration_test rclcpp px4_msgs)
add_executable(sitl_probe test/sitl_probe.cpp)
target_link_libraries(sitl_probe "${typesupport}")
target_include_directories(sitl_probe PRIVATE src)
ament_target_dependencies(sitl_probe rclcpp px4_msgs)
foreach(target snapshot safety_bridge bridge_integration_test sitl_probe)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
install(TARGETS safety_bridge bridge_integration_test sitl_probe DESTINATION lib/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  add_test(NAME ros_safety_bridge_integration
    COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/../../../scripts/test-safety-bridge.sh")
  set_tests_properties(ros_safety_bridge_integration PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
